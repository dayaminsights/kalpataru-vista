  // Elements are NOT cached at setup time: these are React-owned (the hero
  // logo/composite), and hydration replaces the SSR-rendered nodes with new
  // ones shortly after this script's first pass runs (see this file's own
  // "hydration overwrites HTML" rule) -- a cached reference goes stale and
  // silently stops matching what's actually on screen. Re-querying inside
  // sync() every tick costs nothing and is always correct.
  // Hooked to gsap.ticker rather than 'scroll'/rAF/setInterval because the
  // hero's ScrollTrigger scrub recomputes opacity on GSAP's own ticker, and
  // getComputedStyle only reflects that recomputation reliably inside the
  // same tick (local-server.js's probe works around the identical issue by
  // driving gsap.ticker.tick() manually under headless --virtual-time-budget).
  function sync(){
    var heroBg=document.querySelector('.hero_bg__S_r_n');
    var comp=document.querySelector('.hero_composite__3blHB');
    if(!heroBg||!comp)return;
    // Driven by the composite (solid black fill) alone, NOT the stroke-outline
    // layer -- the halo backs the letters only once they read as solid ink.
    // Squared so it eases in: the fill's own scrub curve is fast out of the
    // gate (0 -> 0.48 across the first 100px of its ramp, measured via probe),
    // which snapped the halo on. x*x counteracts that, letting the glow gather
    // gradually while still reaching full strength exactly when the fill does.
    var fill=parseFloat(getComputedStyle(comp).opacity)||0;
    heroBg.style.setProperty('--kv-wordmark-opacity',String(fill*fill));

    gsap.ticker.add(sync);
    sync();
